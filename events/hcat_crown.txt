# HoI IV - Catalunya: the contest for the Crown of Arago.
# Fired by hcat_start_the_crown_contest while CAT_the_count_of_barcelona is in progress.
# Each option adds support to one or more houses (hcat_claim_<house>); the focus crowns
# the leader on completion (hcat_crown_the_leading_claimant). ROOT is Catalonia.
add_namespace = hcat_crown

# Envoys of Juan de Borbon arrive in Barcelona. Queued for day 3 of the focus.
country_event = {
	id = hcat_crown.1
	title = hcat_crown.1.t
	desc = hcat_crown.1.desc
	picture = GFX_report_event_generic_sign_treaty1
	is_triggered_only = yes

	trigger = {
		focus_progress = { focus = CAT_the_count_of_barcelona progress > 0 }
		NOT = { has_country_flag = hcat_crown_event_1_seen }
	}

	immediate = { hidden_effect = { set_country_flag = hcat_crown_event_1_seen } }

	option = {
		name = hcat_crown.1.a
		ai_chance = { base = 3 }
		add_to_variable = { hcat_claim_bourbon = 3 tooltip = hcat_claim_bourbon_tt }
	}

	option = {
		name = hcat_crown.1.b
		ai_chance = { base = 3 }
		add_to_variable = { hcat_claim_barcelona = 2 tooltip = hcat_claim_barcelona_tt }
		add_to_variable = { hcat_claim_urgell = 1 tooltip = hcat_claim_urgell_tt }
	}
}
