# The Corts Have Chosen. Fired by hcat_crown_the_leading_claimant when

	id = hcat_catalonia.4
	title = hcat_catalonia.4.t
	desc = { text = hcat_catalonia.4.desc_bourbon trigger = { has_country_flag = hcat_crown_winner_bourbon } }
	desc = { text = hcat_catalonia.4.desc_barcelona trigger = { has_country_flag = hcat_crown_winner_barcelona } }
	desc = { text = hcat_catalonia.4.desc_habsburg trigger = { has_country_flag = hcat_crown_winner_habsburg } }
	desc = { text = hcat_catalonia.4.desc_urgell trigger = { has_country_flag = hcat_crown_winner_urgell } }
	picture = GFX_report_event_monarchy_question
	is_triggered_only = yes

		name = hcat_catalonia.4.a
		trigger = { has_country_flag = hcat_crown_winner_bourbon }

	}

		name = hcat_catalonia.4.b

		name = hcat_catalonia.4.d

		hcat_crown_house_of_urgell = yes
	}
}

# The civil war has begun: the status quo Catalonia chooses a side.
# Fired by hcat_catalonia_daily_civil_war_check. ROOT is Catalonia.
country_event = {
	id = hcat_catalonia.5
	title = hcat_catalonia.5.t
	desc = hcat_catalonia.5.desc
	picture = GFX_report_event_gathering_protest
	is_triggered_only = yes

	# Loyal to the Republic.
	option = {
		name = hcat_catalonia.5.a
		ai_chance = { base = 5 }
		set_country_flag = hcat_scw_side_republic
		custom_effect_tooltip = hcat_side_with_the_republic_tt
		hidden_effect = { hcat_catalonia_follow_the_republic = yes }
		add_to_variable = { global.hcat_leverage = 5 tooltip = hcat_leverage_change_tt }
		hcat_clamp_leverage = yes
	}

	# Armed neutrality.
	option = {
		name = hcat_catalonia.5.b
		ai_chance = { base = 2 }
		set_country_flag = hcat_scw_side_neutral
		custom_effect_tooltip = hcat_declare_neutrality_tt
		hcat_catalonia_declare_neutrality = yes
	}

	# With the Carlists, who promise the old fueros. Only while they are in arms.
	option = {
		name = hcat_catalonia.5.c
		trigger = {
			any_country = {
				hcat_is_carlist_spain = yes
				has_civil_war = yes
			}
		}
		ai_chance = { base = 1 }
		custom_effect_tooltip = hcat_join_the_carlists_tt
		hidden_effect = { hcat_catalonia_follow_the_carlists = yes }
	}
}
