# HoI IV - Catalunya: events fired to Spain.
add_namespace = hcat_spain

# Catalonia proclaims the Catalan State. ROOT is Spain, FROM is Catalonia.
country_event = {
	id = hcat_spain.1
	title = hcat_spain.1.t
	desc = hcat_spain.1.desc
	picture = GFX_report_event_gathering_protest
	is_triggered_only = yes

	# Let Catalonia go.
	option = {
		name = hcat_spain.1.a
		ai_chance = {
			base = 1
			modifier = {
				factor = 5
				has_civil_war = yes
			}
			modifier = {
				factor = 3
				has_stability < 0.3
			}
		}
		end_puppet = FROM
		add_stability = -0.05
		FROM = { hcat_catalonia_breaks_free = yes }
		set_global_flag = hcat_outcome_independence
		hcat_settle_catalan_question = yes
		FROM = { country_event = { id = hcat_catalonia.1 } }
	}

	# Crush the separatists.
	option = {
		name = hcat_spain.1.b
		ai_chance = { base = 5 }
		end_puppet = FROM
		FROM = { hcat_catalonia_breaks_free = yes }
		set_global_flag = hcat_war_of_catalan_independence
		add_war_support = 0.05
		declare_war_on = {
			target = FROM
			type = annex_everything
		}
		FROM = { country_event = { id = hcat_catalonia.2 } }
	}
}

# Catalonia demands The Federal Republic (CAT_the_federal_republic).
# ROOT is Spain, FROM is Catalonia.
country_event = {
	id = hcat_spain.2
	title = hcat_spain.2.t
	desc = hcat_spain.2.desc
	picture = GFX_report_event_generic_parliament
	is_triggered_only = yes

	# Accept: Catalonia becomes a Federated State.
	option = {
		name = hcat_spain.2.a
		ai_chance = {
			base = 2
			modifier = {
				factor = 3
				check_variable = { global.hcat_leverage > 79 }
			}
			modifier = {
				factor = 2
				has_civil_war = yes
			}
		}
		set_autonomy = {
			target = FROM
			autonomous_state = hcat_autonomy_federated_state
			end_wars = no
			end_civil_wars = no
		}
		add_ideas = hcat_a_federal_spain
		hcat_catalonia_becomes_a_federated_state = yes
		FROM = { country_event = { id = hcat_catalonia.6 } }
	}

	# Refuse: Catalonia rises.
	option = {
		name = hcat_spain.2.b
		ai_chance = { base = 3 }
		end_puppet = FROM
		FROM = { hcat_catalonia_breaks_free = yes }
		set_global_flag = hcat_war_of_federation
		add_stability = -0.10
		declare_war_on = {
			target = FROM
			type = annex_everything
		}
		FROM = { country_event = { id = hcat_catalonia.7 } }
	}
}

# The civil war is over and Catalonia sat it out (hcat_scw_side_neutral).
# ROOT is the winning Spain; Catalonia is addressed as CAT.
country_event = {
	id = hcat_spain.3
	title = hcat_spain.3.t
	desc = hcat_spain.3.desc
	picture = GFX_report_event_generic_sign_treaty1
	is_triggered_only = yes

	trigger = {
		CAT = {
			exists = yes
			is_subject = no
		}
	}

	# Welcome Catalonia back as an Autonomous Community.
	option = {
		name = hcat_spain.3.a
		ai_chance = { base = 3 }
		puppet = CAT
		set_autonomy = {
			target = CAT
			autonomous_state = hcat_autonomy_autonomous_community
		}
		CAT = {
			remove_ideas = hcat_armed_neutrality
			country_event = { id = hcat_catalonia.9 }
		}
	}

	# Neutrality was treason.
	option = {
		name = hcat_spain.3.b
		ai_chance = { base = 1 }
		add_war_support = 0.05
		declare_war_on = {
			target = CAT
			type = annex_everything
		}
		CAT = { country_event = { id = hcat_catalonia.10 } }
	}
}
