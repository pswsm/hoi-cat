# HoI IV - Catalunya: on_actions
on_actions = {
	# on_startup has no scope; borrow any country to set global state.
	on_startup = {
		effect = {
			random_country = { hcat_init = yes }
		}
	}

	# There is no on_action for a civil war starting, so Catalonia checks daily.
	# The check is a few cheap triggers on a single country.
	on_daily_CAT = {
		effect = {
			hcat_catalonia_follow_the_republic = yes
		}
	}

	on_monthly_CAT = {
		effect = {
			hcat_spanish_weakness_drift = yes
		}
	}

	# ROOT is the winner, FROM the loser, which still exists at this point.
	on_civil_war_end_before_annexation = {
		effect = {
			hcat_catalonia_after_the_civil_war = yes
		}
	}

	on_monthly = {
		effect = {
			if = {
				limit = { has_dynamic_modifier = { modifier = hcat_catalan_discontent_country } }
				hcat_fade_catalan_discontent = yes
			}
		}
	}
}
