# HoI IV - Catalunya: on_actions
on_actions = {
	# on_startup has no scope; borrow any country to set global state.
	on_startup = {
		effect = {
			random_country = { hcat_init = yes }
		}
	}

	# There is no on_action for a civil war starting, so Catalonia checks daily.
	# The check is a few cheap triggers on a single country.
	on_daily_CAT = {
		effect = {
			hcat_catalonia_follow_the_republic = yes
		}
	}
}
