# HoI IV - Catalunya: scripted effects
#
# Core state of the mod: global variable `hcat_autonomy` (0-100).
#   0   = Catalonia fully integrated into a centralised Spain
#   50  = the 1932 Statute of Autonomy (game start)
#   100 = de facto independence
# Spanish-side content pushes it down, Catalan-side content pushes it up.
# Global flags and variables still need a scope to run in; any country will do.

hcat_init = {
	if = {
		limit = { NOT = { has_global_flag = hcat_initialised } }
		set_global_flag = hcat_initialised
		set_variable = { global.hcat_autonomy = 50 }
		set_variable = { global.hcat_autonomy_floor = 0 }
	}
}

# HoI4 scripted effects take no parameters, so change the track with:
#   add_to_variable = { global.hcat_autonomy = <n> tooltip = hcat_autonomy_change_tt }
#   hcat_clamp_autonomy = yes
# Also enforces global.hcat_autonomy_floor, set by the Catalan status quo path.
hcat_clamp_autonomy = {
	hidden_effect = {
		clamp_variable = {
			var = global.hcat_autonomy
			min = 0
			max = 100
		}
		if = {
			limit = { check_variable = { global.hcat_autonomy < global.hcat_autonomy_floor } }
			set_variable = { global.hcat_autonomy = global.hcat_autonomy_floor }
		}
	}
}

### Autonomy floor: the Statute guaranteed by the status quo path ###

hcat_raise_autonomy_floor_to_30 = {
	custom_effect_tooltip = hcat_autonomy_floor_30_tt
	hidden_effect = {
		if = {
			limit = { check_variable = { global.hcat_autonomy_floor < 30 } }
			set_variable = { global.hcat_autonomy_floor = 30 }
		}
	}
	hcat_clamp_autonomy = yes
}

hcat_raise_autonomy_floor_to_40 = {
	custom_effect_tooltip = hcat_autonomy_floor_40_tt
	hidden_effect = {
		if = {
			limit = { check_variable = { global.hcat_autonomy_floor < 40 } }
			set_variable = { global.hcat_autonomy_floor = 40 }
		}
	}
	hcat_clamp_autonomy = yes
}

# Country scope: Spain. Spanish recentralisation tears up the Statute guarantee.
hcat_break_the_statute_pact = {
	if = {
		limit = { check_variable = { global.hcat_autonomy_floor > 0 } }
		custom_effect_tooltip = hcat_break_the_statute_pact_tt
		hidden_effect = {
			set_variable = { global.hcat_autonomy_floor = 0 }
			set_global_flag = hcat_statute_pact_broken
		}
		CAT = { add_stability = -0.05 }
		hidden_effect = {
			CAT = { country_event = { id = hcat_catalonia.3 } }
		}
	}
}

# Set the outcome flag first: hcat_outcome_integration, hcat_outcome_status_quo
# or hcat_outcome_independence. Then call hcat_settle_catalan_question = yes.
hcat_settle_catalan_question = {
	set_global_flag = hcat_catalan_question_settled
}

### Raising autonomy from the Catalan side ###

# Country scope: Catalonia. Runs monthly: a weak or divided Spain loses its grip.
hcat_spanish_weakness_drift = {
	if = {
		limit = {
			is_subject = yes
			OVERLORD = { hcat_is_spain = yes }
			hcat_catalan_question_settled = no
		}
		if = {
			limit = { OVERLORD = { has_civil_war = yes } }
			add_to_variable = { global.hcat_autonomy = 2 }
		}
		if = {
			limit = { OVERLORD = { has_stability < 0.3 } }
			add_to_variable = { global.hcat_autonomy = 1 }
		}
		hcat_clamp_autonomy = yes
	}
}

# Country scope: Catalonia. Proclaims the Catalan State; Spain decides what happens.
hcat_proclaim_the_catalan_state = {
	set_global_flag = hcat_catalan_state_proclaimed
	custom_effect_tooltip = hcat_proclaim_the_catalan_state_tt
	hcat_raise_the_catalan_forces = yes
	OVERLORD = { country_event = { id = hcat_spain.1 } }
}

# Country scope: Catalonia, when it proclaims the Catalan State.
# Unlocks militia and military police, then loads history/units/hcat_CAT_catalan_forces.txt
# (templates and divisions live there: OOB files may be UTF-8 with BOM, which the
# accented names need; scripted effect files may not).
hcat_raise_the_catalan_forces = {
	custom_effect_tooltip = hcat_raise_the_catalan_forces_tt
	hidden_effect = {
		set_technology = { militia_tech = 1 tech_military_police = 1 popup = no }
		load_oob = "hcat_CAT_catalan_forces"
	}
}

# Country scope: Catalonia, after it breaks free (either way).
hcat_catalonia_breaks_free = {
	set_variable = { global.hcat_autonomy = 100 }
	set_variable = { global.hcat_autonomy_floor = 0 }
}

# Country scope: Catalonia.
# The Generalitat stays loyal to the Republic. If Spain splits in a civil war and
# Catalonia's overlord ends up on the non-Republican side, its subject relationship
# moves to the Republican side, and Catalonia joins the war against its former overlord.
hcat_catalonia_follow_the_republic = {
	if = {
		limit = {
			is_subject = yes
			OVERLORD = {
				hcat_is_spain = yes
				hcat_is_republican_spain = no
				has_civil_war = yes
			}
		}
		OVERLORD = { save_event_target_as = hcat_former_overlord }
		random_country = {
			limit = {
				hcat_is_republican_spain = yes
				has_war_with = event_target:hcat_former_overlord
			}
			save_event_target_as = hcat_republican_spain
		}
		if = {
			limit = { has_event_target = hcat_republican_spain }
			event_target:hcat_former_overlord = { end_puppet = ROOT }
			event_target:hcat_republican_spain = {
				set_autonomy = {
					target = ROOT
					autonomous_state = hcat_autonomy_autonomous_community
					end_wars = no
					end_civil_wars = no
				}
			}
			add_to_war = {
				targeted_alliance = event_target:hcat_republican_spain
				enemy = event_target:hcat_former_overlord
				hostility_reason = asked_to_join
			}
			set_global_flag = hcat_catalonia_sided_with_the_republic
		}
	}
}

# Country scope: Spain. Annexes Catalonia and starts Catalan discontent.
# Discontent starts at 30 + 2x the remaining autonomy (30-80) and fades by 1 a month.
hcat_integrate_catalonia = {
	set_variable = { global.hcat_discontent = global.hcat_autonomy }
	multiply_variable = { global.hcat_discontent = 2 }
	add_to_variable = { global.hcat_discontent = 30 }
	set_variable = { global.hcat_autonomy = 0 }

	annex_country = {
		target = CAT
		transfer_troops = yes
	}

	hcat_update_catalan_discontent = yes
	add_dynamic_modifier = { modifier = hcat_catalan_discontent_country }
	165 = { add_dynamic_modifier = { modifier = hcat_catalan_discontent_state } }

	set_global_flag = hcat_outcome_integration
	hcat_settle_catalan_question = yes
}

# Country scope: the country holding hcat_catalan_discontent_country.
# Recomputes the modifier values from global.hcat_discontent.
hcat_update_catalan_discontent = {
	set_variable = { hcat_discontent_stability = global.hcat_discontent }
	multiply_variable = { hcat_discontent_stability = -0.001 }
	165 = {
		set_variable = { hcat_discontent_factories = global.hcat_discontent }
		multiply_variable = { hcat_discontent_factories = -0.004 }
		set_variable = { hcat_discontent_recruitable = global.hcat_discontent }
		multiply_variable = { hcat_discontent_recruitable = -0.004 }
	}
}

# Country scope: the country holding hcat_catalan_discontent_country. Runs monthly.
hcat_fade_catalan_discontent = {
	subtract_from_variable = { global.hcat_discontent = 1 }
	if = {
		limit = { check_variable = { global.hcat_discontent < 1 } }
		set_variable = { global.hcat_discontent = 0 }
		remove_dynamic_modifier = { modifier = hcat_catalan_discontent_country }
		165 = { remove_dynamic_modifier = { modifier = hcat_catalan_discontent_state } }
	}
	else = {
		hcat_update_catalan_discontent = yes
	}
}

# Country scope: ROOT is the civil war winner, FROM the loser about to be annexed.
# A Spanish winner takes over Catalonia from the losing side instead of it walking
# free. Nationalists can then abolish the Statute through their focus branch.
hcat_catalonia_after_the_civil_war = {
	if = {
		limit = {
			hcat_is_spain = yes
			CAT = {
				exists = yes
				is_subject_of = FROM
			}
		}
		FROM = { end_puppet = CAT }
		set_autonomy = {
			target = CAT
			autonomous_state = hcat_autonomy_autonomous_community
		}
		if = {
			limit = { hcat_is_republican_spain = no }
			set_global_flag = hcat_catalonia_lost_the_civil_war
		}
	}
}

# Country scope: Catalonia, from each hcat_crown_house_of_<house>, after
# recruit_character and before promote_character.
hcat_crown_the_claimant = {
	set_politics = { ruling_party = neutrality elections_allowed = no }
	add_popularity = { ideology = neutrality popularity = 0.20 }
	# Principat de Catalunya (flag: gfx/flags/CAT_principality.tga). The Kingdom of Aragon
	# comes later, from hcat_proclaim_the_kingdom_of_aragon.
	set_cosmetic_tag = CAT_principality
}

### The contest for the Crown of Arago (CAT_the_count_of_barcelona, 35 days) ###
# Support is held in four country variables on Catalonia: hcat_claim_<house>.

# Country scope: Catalonia, from the focus's select_effect.
# Every hcat_crown event is queued now; each one only fires while the focus is still in
# progress and only once per contest (hcat_crown_event_<n>_seen), so cancelling or
# restarting the focus does not double them up.
hcat_start_the_crown_contest = {
	set_variable = { hcat_claim_bourbon = 4 }
	set_variable = { hcat_claim_barcelona = 4 }
	set_variable = { hcat_claim_habsburg = 2 }
	set_variable = { hcat_claim_urgell = 2 }
	clr_country_flag = hcat_crown_event_1_seen
	clr_country_flag = hcat_crown_event_2_seen
	clr_country_flag = hcat_crown_event_3_seen
	clr_country_flag = hcat_crown_event_4_seen
	clr_country_flag = hcat_crown_event_5_seen
	country_event = { id = hcat_crown.1 days = 3 }
	country_event = { id = hcat_crown.2 days = 10 }
	country_event = { id = hcat_crown.3 days = 17 }
	country_event = { id = hcat_crown.4 days = 24 }
	country_event = { id = hcat_crown.5 days = 31 }
}

# Country scope: Catalonia, when the focus completes. Crowns the house with the most
# support. Ties go, in order, to Barcelona, Bourbon, Habsburg, Urgell.
hcat_crown_the_leading_claimant = {
	if = {
		limit = {
			NOT = { check_variable = { hcat_claim_barcelona < hcat_claim_bourbon } }
			NOT = { check_variable = { hcat_claim_barcelona < hcat_claim_habsburg } }
			NOT = { check_variable = { hcat_claim_barcelona < hcat_claim_urgell } }
		}
		set_country_flag = hcat_crown_winner_barcelona
	}
	else_if = {
		limit = {
			NOT = { check_variable = { hcat_claim_bourbon < hcat_claim_habsburg } }
			NOT = { check_variable = { hcat_claim_bourbon < hcat_claim_urgell } }
		}
		set_country_flag = hcat_crown_winner_bourbon
	}
	else_if = {
		limit = {
			NOT = { check_variable = { hcat_claim_habsburg < hcat_claim_urgell } }
		}
		set_country_flag = hcat_crown_winner_habsburg
	}
	else = {
		set_country_flag = hcat_crown_winner_urgell
	}
	country_event = { id = hcat_catalonia.4 }
}
